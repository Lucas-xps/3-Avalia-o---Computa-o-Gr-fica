<script>
    (function() {
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const pointsList = document.getElementById('pointsList');
        const badgeCount = document.getElementById('badgeCount');
        const degreeLabel = document.getElementById('degreeLabel');
        const pointCountLabel = document.getElementById('pointCountLabel');
        const statusMsg = document.getElementById('statusMsg');
        const precision = document.getElementById('precision');
        const precisionDisplay = document.getElementById('precisionDisplay');

        const drawBtn = document.getElementById('drawBtn');
        const randomBtn = document.getElementById('randomBtn');
        const resetBtn = document.getElementById('resetBtn');
        const curveColorInput = document.getElementById('curveColor');

        // Pontos de controle (P0, P1, P2, P3)
        let pts = [
            { x: 60, y: 320 },  // P0 - início
            { x: 140, y: 80 },  // P1 - controle
            { x: 300, y: 100 }, // P2 - controle
            { x: 340, y: 300 }  // P3 - fim (apenas grau 3)
        ];

        const cores = ['#c0392b', '#2980b9', '#27ae60', '#8e44ad'];
        const nomes = ['Início', 'Controle 1', 'Controle 2', 'Fim'];
        let corCurva = '#e91e63';

        function getGrau() {
            return parseInt(document.querySelector('input[name="degree"]:checked').value, 10);
        }

        function getPrecisao() {
            return parseInt(precision.value, 10) || 400;
        }

        // ---- Interface ----
        function renderizarLista() {
            const grau = getGrau();
            const count = grau === 2 ? 3 : 4;
            let html = '';
            for (let i = 0; i < count; i++) {
                html += `
                    <div class="point-item" data-idx="${i}">
                        <span class="idx p${i}">#${i+1}</span>
                        <span class="label-name">${nomes[i]}</span>
                        <input type="number" class="pt-x" data-idx="${i}" value="${pts[i].x}" min="0" max="399" step="1">
                        <input type="number" class="pt-y" data-idx="${i}" value="${pts[i].y}" min="0" max="399" step="1">
                    </div>
                `;
            }
            pointsList.innerHTML = html;
            badgeCount.textContent = count;
            pointCountLabel.textContent = count;
            degreeLabel.textContent = `Grau ${grau}`;

            document.querySelectorAll('.pt-x').forEach((inp) => {
                inp.addEventListener('input', () => {
                    const idx = parseInt(inp.dataset.idx, 10);
                    const val = parseInt(inp.value, 10);
                    if (!isNaN(val) && val >= 0 && val < 400) {
                        pts[idx].x = val;
                        desenhar();
                    }
                });
            });
            document.querySelectorAll('.pt-y').forEach((inp) => {
                inp.addEventListener('input', () => {
                    const idx = parseInt(inp.dataset.idx, 10);
                    const val = parseInt(inp.value, 10);
                    if (!isNaN(val) && val >= 0 && val < 400) {
                        pts[idx].y = val;
                        desenhar();
                    }
                });
            });
        }

        // ---- Rasterização ----
        function putPixel(x, y, cor = corCurva) {
            if (x < 0 || x >= 400 || y < 0 || y >= 400) return;
            const img = ctx.getImageData(0, 0, 400, 400);
            const d = img.data;
            const idx = (y * 400 + x) * 4;
            const r = parseInt(cor.slice(1,3), 16);
            const g = parseInt(cor.slice(3,5), 16);
            const b = parseInt(cor.slice(5,7), 16);
            d[idx] = r; d[idx+1] = g; d[idx+2] = b; d[idx+3] = 255;
            ctx.putImageData(img, 0, 0);
        }

        function bresenham(x0, y0, x1, y1) {
            const dx = Math.abs(x1 - x0), dy = Math.abs(y1 - y0);
            const sx = x0 < x1 ? 1 : -1, sy = y0 < y1 ? 1 : -1;
            let err = dx - dy;
            while (true) {
                putPixel(x0, y0);
                if (x0 === x1 && y0 === y1) break;
                const e2 = 2 * err;
                if (e2 > -dy) { err -= dy; x0 += sx; }
                if (e2 < dx) { err += dx; y0 += sy; }
            }
        }

        function desenharCurva(grau) {
            const n = grau;
            const passos = getPrecisao();
            const pontosCurva = [];

            function binomial(n, k) {
                if (k < 0 || k > n) return 0;
                let r = 1;
                for (let i = 1; i <= k; i++) r *= (n - i + 1) / i;
                return r;
            }

            for (let i = 0; i <= passos; i++) {
                const t = i / passos;
                let x = 0, y = 0;
                for (let j = 0; j <= n; j++) {
                    const coef = binomial(n, j) * Math.pow(t, j) * Math.pow(1 - t, n - j);
                    x += pts[j].x * coef;
                    y += pts[j].y * coef;
                }
                pontosCurva.push({ x: Math.round(x), y: Math.round(y) });
            }

            for (let i = 0; i < pontosCurva.length - 1; i++) {
                bresenham(pontosCurva[i].x, pontosCurva[i].y, pontosCurva[i+1].x, pontosCurva[i+1].y);
            }
        }

        function desenharPontos(grau) {
            const count = grau === 2 ? 3 : 4;
            for (let i = 0; i < count; i++) {
                const p = pts[i];
                const cor = cores[i];
                for (let dy = -3; dy <= 3; dy++) {
                    for (let dx = -3; dx <= 3; dx++) {
                        putPixel(p.x + dx, p.y + dy, cor);
                    }
                }
                // Rótulo do ponto
                ctx.fillStyle = cor;
                ctx.font = '8px monospace';
                ctx.textAlign = 'center';
                // Usa fillText diretamente no canvas (não via putPixel)
                const img = ctx.getImageData(0, 0, 400, 400);
                const d = img.data;
                const label = `${i+1}`;
                const x = p.x + 8, y = p.y - 8;
                for (let dy2 = 0; dy2 < 8; dy2++) {
                    for (let dx2 = 0; dx2 < 8; dx2++) {
                        const px = x + dx2, py = y + dy2;
                        if (px >= 0 && px < 400 && py >= 0 && py < 400) {
                            const idx2 = (py * 400 + px) * 4;
                            if (idx2 < d.length) {
                                d[idx2] = 200; d[idx2+1] = 200; d[idx2+2] = 200; d[idx2+3] = 200;
                            }
                        }
                    }
                }
                ctx.putImageData(img, 0, 0);
            }
        }

        function desenharLinhasGuia(grau) {
            const count = grau === 2 ? 3 : 4;
            const img = ctx.getImageData(0, 0, 400, 400);
            const d = img.data;
            // Desenha linhas guia pixel a pixel
            function putPixelGuide(x, y) {
                if (x < 0 || x >= 400 || y < 0 || y >= 400) return;
                const idx = (y * 400 + x) * 4;
                if (idx < d.length) {
                    d[idx] = 100; d[idx+1] = 120; d[idx+2] = 140; d[idx+3] = 180;
                }
            }
            function bresenhamGuide(x0, y0, x1, y1) {
                const dx = Math.abs(x1 - x0), dy = Math.abs(y1 - y0);
                const sx = x0 < x1 ? 1 : -1, sy = y0 < y1 ? 1 : -1;
                let err = dx - dy;
                while (true) {
                    putPixelGuide(x0, y0);
                    if (x0 === x1 && y0 === y1) break;
                    const e2 = 2 * err;
                    if (e2 > -dy) { err -= dy; x0 += sx; }
                    if (e2 < dx) { err += dx; y0 += sy; }
                }
            }
            for (let i = 0; i < count - 1; i++) {
                bresenhamGuide(pts[i].x, pts[i].y, pts[i+1].x, pts[i+1].y);
            }
            ctx.putImageData(img, 0, 0);
        }

        function limpar() {
            ctx.fillStyle = '#0f0f1f';
            ctx.fillRect(0, 0, 400, 400);
            ctx.strokeStyle = '#1a1a3a';
            ctx.lineWidth = 0.5;
            for (let i = 0; i <= 400; i += 40) {
                ctx.beginPath();
                ctx.moveTo(i, 0);
                ctx.lineTo(i, 400);
                ctx.stroke();
                ctx.beginPath();
                ctx.moveTo(0, i);
                ctx.lineTo(400, i);
                ctx.stroke();
            }
        }

        function desenhar() {
            const grau = getGrau();
            limpar();
            desenharLinhasGuia(grau);
            desenharCurva(grau);
            desenharPontos(grau);
            statusMsg.textContent = `✅ Curva de grau ${grau} desenhada com ${getPrecisao()} passos`;
        }

        // ---- Gerenciamento ----
        function gerarAleatorio() {
            const grau = getGrau();
            const count = grau === 2 ? 3 : 4;
            for (let i = 0; i < count; i++) {
                pts[i] = {
                    x: Math.floor(Math.random() * 380) + 10,
                    y: Math.floor(Math.random() * 380) + 10
                };
            }
            renderizarLista();
            desenhar();
            statusMsg.innerHTML = `🎲 Pontos aleatórios <span style="color:#2ecc71;">gerados</span>!`;
        }

        function resetarPontos() {
            pts = [
                { x: 60, y: 320 },
                { x: 140, y: 80 },
                { x: 300, y: 100 },
                { x: 340, y: 300 }
            ];
            renderizarLista();
            desenhar();
            statusMsg.textContent = '↺ Pontos resetados para o padrão';
        }

        // ---- Eventos ----
        document.querySelectorAll('input[name="degree"]').forEach(radio => {
            radio.addEventListener('change', () => {
                renderizarLista();
                desenhar();
            });
        });

        drawBtn.addEventListener('click', desenhar);
        randomBtn.addEventListener('click', gerarAleatorio);
        resetBtn.addEventListener('click', resetarPontos);

        curveColorInput.addEventListener('input', () => {
            corCurva = curveColorInput.value;
            desenhar();
        });

        precision.addEventListener('input', () => {
            precisionDisplay.textContent = precision.value;
        });

        // Clique no canvas para mover pontos
        canvas.addEventListener('click', (e) => {
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / rect.width;
            const scaleY = canvas.height / rect.height;
            const mx = Math.round((e.clientX - rect.left) * scaleX);
            const my = Math.round((e.clientY - rect.top) * scaleY);
            if (mx < 0 || mx >= 400 || my < 0 || my >= 400) return;

            const grau = getGrau();
            const count = grau === 2 ? 3 : 4;
            let idx = 0, minDist = Infinity;
            for (let i = 0; i < count; i++) {
                const d = (mx - pts[i].x) ** 2 + (my - pts[i].y) ** 2;
                if (d < minDist) { minDist = d; idx = i; }
            }
            if (minDist < 2500) {
                pts[idx].x = mx;
                pts[idx].y = my;
                renderizarLista();
                desenhar();
                statusMsg.textContent = `📍 Ponto ${nomes[idx]} movido para (${mx}, ${my})`;
            }
        });

        // Inicialização
        renderizarLista();
        desenhar();
        statusMsg.textContent = '🖱️ Clique no canvas para mover os pontos de controle';

        window.__state = { pts };
